Quickstart

Requires .NET 10.

shell 
git clone https://github.com/Neftedollar/ll-lang.git
cd ll-lang && dotnet build
hello.lll 
module Hello

greet(name Str) = printfn "Hello, %s!" name
Hello = greet "ll-lang"
shell 
lllc run hello.lll
# Hello, ll-lang!

CLI reference

shell 
lllc build <file.lll>               # compile to F# (default)
lllc build --target ts <file.lll>   # compile to TypeScript
lllc build --target py <file.lll>   # compile to Python
lllc build --target java <file.lll> # compile to Java 21
lllc build --target cs <file.lll>   # compile to C#
lllc build --target llvm <file.lll> # compile to LLVM IR (experimental)
lllc build [dir]                    # compile project (reads lll.toml)
lllc check <file.lll>               # type-check only, no codegen
lllc run   <file.lll>               # compile and run via temp fsproj
lllc new   <name>                   # scaffold new project
lllc install                        # resolve deps into vendor/
lllc mod tidy                       # sync deps
lllc mod add dep=https://repo#ref   # add a dependency
lllc mod why dep                    # explain dependency chain
lllc mcp                            # run MCP server (stdio)

Project setup

Multi-file projects use a lll.toml manifest.

lll.toml 
[project]
name = "myapp"

[platform]
use = ["fsharp", "typescript"]   # emit both targets on lllc build

Syntax overview

Uppercase names declare types; lowercase names declare values. No fn, type, in, then, or with keywords. Layout-sensitive (significant indentation).

ll-lang 
module Examples.Basics

pi = 3.14159

add(a Int)(b Int) Int = a + b
double(x Int) = x * 2

clamp(x Int)(lo Int)(hi Int) Int =
  if x < lo
    lo
  else if x > hi
    hi
  else x

triple = \x. x * 3

example =
  y = double 5
  y + 1

ADTs & pattern matching

ll-lang 
module Examples.ADTs

Shape = Circle Float | Rect Float Float | Empty

Maybe A = Some A | None
Result A E = Ok A | Err E

area(s Shape) Float =
  match s
    | Circle r -> 3.14159 * r * r
    | Rect w h -> w * h
    | Empty    -> 0.0

safeDivide(a Float)(b Float) Maybe[Float] =
  if b == 0.0 None
  else Some (a / b)

Traits

ll-lang 
module Examples.Traits

trait Show A =
  show(a A) Str

impl Show Int =
  show(n Int) Str = intToStr n

impl Show Bool =
  show(b Bool) Str =
    if b "true"
    else "false"

printVal(x A) [Show A] = printfn (show x)

Tags & unit algebra

Tags are zero-cost type wrappers preventing confusion between values of the same base type.

ll-lang 
module Examples.Tags

tag UserId
tag Email

uid = "user-42"[UserId]

getUser(id Str[UserId])  Maybe[Str] = Some "alice"
sendEmail(to Str[Email])            = to

tag m
tag s

-- inferred return type: Float[m/s]
speed(d Float[m])(t Float[s]) = d / t

Keywords

ll-lang has exactly 15 keywords. Everything else is expressed through the uppercase/lowercase convention.
